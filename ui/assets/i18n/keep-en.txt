# 用途：check-i18n.mjs 反向检查的准入清单。
#       界面文本节点 / title / placeholder / aria-label / alt 属性
#       若「不含中文」，必须精确命中本文件某一行（或匹配某条 re: 正则），
#       否则门控失败。
#
# 准入标准（三条任一）：
#   1. 专有名词：品牌、产品、模型系列、协议名 —— 翻译会破坏识别
#   2. 机器可读值：文件路径、URL、字段名、协议参数 —— 用户要照着敲
#   3. 行业通用缩写：日志级别、单位 —— 中文读者也按原样理解
#
# 不准入：
#   - 界面标签、按钮、标题、说明文字（一律必须中文化，英文界面走字典）
#   - 中文标签旁边的英文装饰（kicker、术语胶囊、BASIC/OPTIONAL 分组标）
#   - 英文界面下会重复的译名（中文标签已有的英文对译）
#
# 改动本文件需在 PR 说明理由。新增一行 = 承认这个位置不翻译。

# ── 正则模式 ──
# 文件路径（绝对 / 家目录 / 相对）
re:^(~|\.)?/[A-Za-z0-9_.\-/]+$
re:^[A-Za-z0-9_.\-]+/[A-Za-z0-9_.\-]+\.(json|toml|yaml|yml|md|js|ts)$
# GitHub 仓库路径（owner/repo，无扩展名）
re:^[A-Za-z0-9_.\-]+/[A-Za-z0-9_.\-]+$
# 裸文件名
re:^[A-Za-z0-9_.\-]+\.(json|toml|yaml|yml|md|js|ts)$
# 带千分位的 token 容量
re:^[\d,]+ tokens$
# 多行参数样例（key=value 换行拼接）
re:^[a-z_]+=[^\n]+(\n[a-z_]+=[^\n]+)*$
# URL 与本地地址
re:^https?://
re:^Base URL:
re:^POST\s
# 数值与耗时
re:^[\d.,]+\s*(ms|s|K|M|%)?$
re:^\d{3}\s+[A-Z]+$
# 占位模板（必须含花括号变量，如 {prefix} {label} ({provider})）
re:^.*\{[A-Za-z]+\}.*$
